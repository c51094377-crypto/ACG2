 );

 scene.add(water);

 const waveHeightRange = document.getElementById('waveHeightRange');

 }

 toggleBoxAttachment();

 if (e.key === '1') {
 activeModes[1] = !activeModes[1];
 if (activeModes[1]) {
 if (bones['旋回']) boneVelocities['旋回'] = { rotX: 0, rotY: 0, rotZ: 0, posX: 0, posY: 0, posZ: 0 };
 if (bones['仰角']) boneVelocities['仰角'] = { rotX: 0, rotY: 0, rotZ: 0, posX: 0, posY: 0, posZ: 0 };

 if (e.key === '2') activeModes[2] = !activeModes[2];
 if (e.key === '3') activeModes[3] = !activeModes[3];

 if (e.key === 'w' || e.key === 'W') keys.w = true;

 window.addEventListener('pointerup', () => {
 isDragging = false;
 });

 window.addEventListener('resize', onWindowResize);

 const fileInput = document.getElementById('fileInput');
 fileInput.addEventListener('change', async (e) => {
 const file = e.target.files[0];
 if (!file) return;

 const buffer = await file.arrayBuffer();
 const unzipped = unzipSync(new Uint8Array(buffer));
 const files = {};

 for (const filename in unzipped) {
 files[filename] = unzipped[filename];
 }

 let taigaPath = '';
 for (const filename in files) {
 if (filename.toLowerCase().endsWith('taiga.glb')) {
 taigaPath = filename;

 }
 }

 if (!taigaPath) {
 alert('ZIP内に TAIGA.glb が見つかりません。');
 return;
 }

 const taigaBlob = new Blob([files[taigaPath].buffer]);
 const taigaUrl = URL.createObjectURL(taigaBlob);

 const loader = new GLTFLoader();
 loader.load(taigaUrl, (gltf) => {
 if (currentModel) {
 scene.remove(currentModel);
 colliders = [];
 fileCamera = null;
 fileCamera2 = null;
 originalCamera2Parent = null;
 playerMesh = null;
 bones = {};
 boneVelocities = {};
 boatsData = [];
 boatCameras = {};
 radarMeshes = [];
 activeBoatKey = null;
 npcShips = [];
 }

 currentModel = gltf.scene;
 scene.add(currentModel);

 currentModel.traverse((child) => {
 if (child.name === 'レーダー' || child.name === 'レーダー_1' || child.name === 'レーダー_2') {
 radarMeshes.push(child);
 }
 if (child.isMesh) {
 colliders.push(child);
 child.frustumCulled = false;
 if (child.material) {
 if (Array.isArray(child.material)) {
 child.material.forEach(m => {
 m.polygonOffset = false;
 m.side = THREE.DoubleSide;
 m.needsUpdate = true;
 });

 child.material.polygonOffset = false;
 child.material.side = THREE.DoubleSide;
 child.material.needsUpdate = true;
 }
 }
 }
 if (child.isBone || child.isObject3D) {
 bones[child.name] = child;
 boneVelocities[child.name] = { rotX: 0, rotY: 0, rotZ: 0, posX: 0, posY: 0, posZ: 0 };
 if (child.name === 'PerspectiveCamera_2') {
 fileCamera2 = child;
 originalCamera2Parent = child.parent;
 }
 if (child.name === 'PerspectiveCamera_4' || child.name === 'PerspectiveCamera_6') {
 boatCameras[child.name] = child;
 }
 }
 });

 let empty6MeshForInit = null;
 currentModel.traverse((child) => {
 if (child.name === 'empty_6' && !empty6MeshForInit) {
 empty6MeshForInit = child;
 }
 });

 currentModel.updateMatrixWorld(true);

 ['ボート', 'ボート2'].forEach(boatName => {
 let boatObj = null;

 }
 });

 if (boatObj) {
 boatObj.traverse((subChild) => {
 if (subChild.isMesh && !colliders.includes(subChild)) {
 colliders.push(subChild);
 subChild.frustumCulled = false;
 if (subChild.material) {
 if (Array.isArray(subChild.material)) {
 subChild.material.forEach(m => {
 m.polygonOffset = false;
 m.side = THREE.DoubleSide;
 m.needsUpdate = true;
 });
 } else {
 subChild.material.polygonOffset = false;
 subChild.material.side = THREE.DoubleSide;
 subChild.material.needsUpdate = true;
 }
 }
 }
 });

 const initialLocalScale = boatObj.scale.clone();

 let isInitiallyTouching = false;
 if (empty6MeshForInit) {
 boatObj.updateMatrixWorld(true);
 empty6MeshForInit.updateMatrixWorld(true);

 const empty6Box = new THREE.Box3().setFromObject(empty6MeshForInit);

 }

 if (isInitiallyTouching && empty6MeshForInit) {

 const worldQuat = new THREE.Quaternion();
 boatObj.getWorldPosition(worldPos);
 boatObj.getWorldQuaternion(worldQuat);

 empty6MeshForInit.attach(boatObj);
 empty6MeshForInit.updateMatrixWorld(true);
 const inverseParentMatrix = new THREE.Matrix4().copy(boatObj.parent.matrixWorld).invert();
 const targetWorldMatrix = new THREE.Matrix4().compose(worldPos, worldQuat, initialLocalScale);
 const localMatrix = targetWorldMatrix.premultiply(inverseParentMatrix);
 localMatrix.decompose(boatObj.position, boatObj.quaternion, boatObj.scale);
 boatObj.updateMatrixWorld(true);
 }

 let camName = boatName === 'ボート' ? 'PerspectiveCamera_4' : 'PerspectiveCamera_6';

 name: boatName,
 object: boatObj,

 function createPlayerMesh() {
 if (playerMesh) return;
 
 const group = new THREE.Group();
 const mat = new THREE.MeshStandardMaterial({ color: 0x3366cc, roughness: 0.5 });
 const skinMat = new THREE.MeshStandardMaterial({ color: 0xffcc99, roughness: 0.5 });

 const head = new THREE.Mesh(new THREE.SphereGeometry(0.18, 16, 16), skinMat);
 head.position.y = 1.45;
 head.castShadow = true;
 group.add(head);

 const torso = new THREE.Mesh(new THREE.CylinderGeometry(0.22, 0.16, 0.6, 16), mat);
 torso.position.y = 1.05;
 torso.castShadow = true;
 group.add(torso);

 const hips = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.18, 0.2, 16), mat);
 hips.position.y = 0.7;
 hips.castShadow = true;
 group.add(hips);

 const armGeo = new THREE.CylinderGeometry(0.07, 0.05, 0.55, 8);
 const leftArm = new THREE.Mesh(armGeo, mat);
 leftArm.position.set(-0.3, 1.05, 0);
 leftArm.castShadow = true;
 group.add(leftArm);

 const rightArm = new THREE.Mesh(armGeo, mat);
 rightArm.position.set(0.3, 1.05, 0);
 rightArm.castShadow = true;
 group.add(rightArm);

 const legGeo = new THREE.CylinderGeometry(0.08, 0.06, 0.7, 8);
 const leftLeg = new THREE.Mesh(legGeo, mat);
 leftLeg.position.set(-0.12, 0.35, 0);
 leftLeg.castShadow = true;
 group.add(leftLeg);

 const rightLeg = new THREE.Mesh(legGeo, mat);
 rightLeg.position.set(0.12, 0.35, 0);
 rightLeg.castShadow = true;
 group.add(rightLeg);

 playerMesh = group;
 }

 function toggleCamera() {
 if (!fileCamera) {
 console.warn("GLBファイル内にカメラが見つかりません。");
 return;
 }

 if (isNavigationCameraActive) {
 isNavigationCameraActive = false;
 if (fileCamera2 && originalCamera2Parent) {
 originalCamera2Parent.attach(fileCamera2);
 }
 }
 if (activeBoatKey) {
 const currentBoat = boatsData.find(b => b.name === activeBoatKey);
 if (currentBoat && currentBoat.camera) {
 currentBoat.camera.parent.attach(currentBoat.camera);
 }
 activeBoatKey = null;
 }

 isFileCameraActive = !isFileCameraActive;
 if (isFileCameraActive) {
 if (originalCameraParent && originalCameraParent !== currentModel) {
 currentModel.attach(fileCamera);
 }

 createPlayerMesh();
 if (playerMesh.parent !== currentModel) {
 currentModel.add(playerMesh);
 }
 playerMesh.visible = true;
 playerMesh.position.copy(fileCamera.position);
 playerMesh.position.y -= 1.688;

 activeCamera = fileCamera;
 controls.enabled = false;
 
 const euler = new THREE.Euler(0, 0, 0, 'YXZ');
 euler.setFromQuaternion(fileCamera.quaternion);
 cameraRotation.yaw = euler.y;
 cameraRotation.pitch = euler.x;
 cameraPhysics.velocityY = 0;
 } else {
 activeCamera = freeCamera;
 controls.enabled = true;
 if (playerMesh) {
 playerMesh.visible = true;
 }
 }
 }

 function toggleNavigationCamera() {
 if (!fileCamera2) {
 console.warn("PerspectiveCamera_2 が見つかりません。");
 return;
 }

 if (isFileCameraActive) {
 isFileCameraActive = false;
 }
 if (activeBoatKey) {
 const currentBoat = boatsData.find(b => b.name === activeBoatKey);
 if (currentBoat && currentBoat.camera) {
 currentBoat.camera.parent.attach(currentBoat.camera);
 }
 activeBoatKey = null;
 }

 isNavigationCameraActive = !isNavigationCameraActive;
 if (isNavigationCameraActive) {
 if (currentModel) {
 currentModel.attach(fileCamera2);
 }

 if (playerMesh) {
 playerMesh.visible = true;
 }

 activeCamera = fileCamera2;
 controls.enabled = false;

 const euler = new THREE.Euler(0, 0, 0, 'YXZ');
 euler.setFromQuaternion(fileCamera2.quaternion);
 cameraRotation.yaw = euler.y;
 cameraRotation.pitch = euler.x;
 } else {
 if (originalCamera2Parent) {
 originalCamera2Parent.attach(fileCamera2);
 }
 activeCamera = freeCamera;
 controls.enabled = true;
 if (playerMesh) {
 playerMesh.visible = true;
 }
 }
 }

 function toggleSpecificBoatCamera(boatName, cameraName) {
 const targetBoat = boatsData.find(b => b.name === boatName);
 const targetCam = boatCameras[cameraName];

 if (!targetBoat || !targetCam) {
 console.warn(`対象のボート (${boatName}) またはカメラ (${cameraName}) が見つかりません。`);
 return;
}

 if (isFileCameraActive) isFileCameraActive = false;

 function calculateWaveHeight(x, y, time) {
 const f = waveParams.frequency;
 const h = waveParams.currentHeight;

 const largeWave = Math.sin(x * f + time * 1.2) * Math.cos(y * f + time * 0.9);
 const smallWave = Math.sin(x * (f * 3.0) - time * 2.5) * Math.cos(y * (f * 2.5) - time * 2.0) * 0.3;

 return (largeWave + smallWave) * h;
 }
